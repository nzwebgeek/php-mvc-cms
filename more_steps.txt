public function create(
    int $userId,
    array $data
): bool {

    $stmt = $this->db->prepare("
        INSERT INTO posts (

            user_id,
            title,
            slug,
            content,
            status,
            created_at,
            updated_at

        ) VALUES (

            :user_id,
            :title,
            :slug,
            :content,
            :status,
            NOW(),
            NOW()

        )
    ");

    return $stmt->execute([

        'user_id' => $userId,
        'title'   => $data['title'],
        'slug'    => $data['slug'],
        'content' => $data['content'],
        'status'  => $data['status']

    ]);
}

Step 2 - Create a controller method

In DashboardController.php, this is added below updatePost():
public function storePost(): void
{
    $user = $this->currentUser();

    $this->posts->create(
        (int)$user['id'],
        $_POST
    );

    $_SESSION['message'] = "Post created successfully.";

    header('Location: /dashboard?panel=posts');

    exit;
}
